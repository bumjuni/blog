#include <iostream>
#include <string>
#include <vector>
#include <unordered_set>
#include <algorithm>

using namespace std;

vector<bool> primes(10000000, true);
vector<bool> visited(7, false);
unordered_set<int> numbers_made;

void sieve(int max) {
    primes[0] = false;
    primes[1] = false;

    for (int i=2; i*i<max; ++i) {
        for (int j=i*i; j<=max; j+=i) {
            primes[j] = false;
        }
    }
}

void make_numbers(string& numbers, string cur, int depth) {
    cout << "cur: " << cur;
    if (cur.size())    numbers_made.insert(stoi(cur));
    for (int i=0; i<numbers.size(); ++i) {
        if (visited[i]) continue;
        if (depth == 1 && numbers[i] == '0') continue;
        cout << " n: " << numbers[i] << endl;
        visited[i] = true;
        make_numbers(numbers, cur+numbers[i], depth+1);
        visited[i] = false;
    }
}

int solution(string numbers) {
    // 1. 숫자 만들기
    make_numbers(numbers, "", 1);

    // 2. 소수 판별하기
    int max = *max_element(numbers_made.begin(), numbers_made.end());
    sieve(max);

    int answer = 0;
    for (int i : numbers_made) {
        if (primes[i]) answer++;
    }
    return answer;
}

int main() {
    cout << solution("011");

    return 0;
}
